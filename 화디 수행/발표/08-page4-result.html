태그 매칭 알고리즘 — 점수가 가장 높은 곡 선택
사용자의 5개 답을 배열로 묶고, songDatabase의 모든 곡과 비교. 일치하는 태그 1개당 +1점. 최고점을 받은 곡을 결과로 반환한다.
▌ 점수 계산 로직 (result.js) 
// 사용자의 5개 답
const userChoices = [weather, mood, companion, time, place];
let bestSong = songDatabase[0], maxScore = -1;
for (let song of songDatabase) {
  let score = 0;
  for (let choice of userChoices) {
    if (song.tags.includes(choice)) score++;   // 일치 +1
  }
  if (score > maxScore) { maxScore = score; bestSong = song; }
}
videoEl.src = "https://www.youtube.com/embed/" + bestSong.youtubeId;
